#### 方法B：Python手动绘制

```python
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# 读取数据
data = pd.read_csv('MetricsData.csv')

# 绘制HV箱线图

sns.boxplot(x='Algorithm', y='HV', data=data)
plt.title('Hypervolume Comparison')

plt.show()

# 绘制IGD箱线图
plt.figure(figsize=(10, 6))
sns.boxplot(x='Algorithm', y='IGD', data=data)
plt.title('IGD Comparison')
plt.savefig('igd_boxplot.png', dpi=300)
plt.show()
```

## 📊 高级用法

### 添加散点

```python
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

data = pd.read_csv('MetricsData.csv')

plt.figure(figsize=(10, 6))
sns.boxplot(x='Algorithm', y='HV', data=data)
sns.stripplot(x='Algorithm', y='HV', data=data, 
              color='black', alpha=0.5, size=4)
plt.title('Hypervolume with Data Points')
plt.savefig('hv_with_points.png', dpi=300)
```

### 小提琴图

```python
plt.figure(figsize=(10, 6))
sns.violinplot(x='Algorithm', y='HV', data=data)
plt.title('Hypervolume Distribution')
plt.savefig('hv_violin.png', dpi=300)
```

### 分组对比

```python
fig, axes = plt.subplots(1, 2, figsize=(15, 6))

# HV
sns.boxplot(x='Algorithm', y='HV', data=data, ax=axes[0])
axes[0].set_title('Hypervolume')

# IGD
sns.boxplot(x='Algorithm', y='IGD', data=data, ax=axes[1])
axes[1].set_title('IGD')

plt.tight_layout()
plt.savefig('combined_boxplot.png', dpi=300)
```

### 统计注释

```python
from scipy import stats

# Wilcoxon检验
nsgaii = data[data['Algorithm']=='NSGAII']['HV']
moead = data[data['Algorithm']=='MOEAD']['HV']
_, p = stats.wilcoxon(nsgaii, moead)

# 在图上标注显著性
plt.figure(figsize=(10, 6))
ax = sns.boxplot(x='Algorithm', y='HV', data=data)

if p < 0.01:
    # 添加 ** 标记
    y_max = data['HV'].max()
    ax.text(0.5, y_max*1.05, '**', ha='center', fontsize=20)
    
plt.title(f'Hypervolume Comparison (p={p:.4f})')
plt.savefig('hv_with_sig.png', dpi=300)
```

## 🎓 完整示例

### 一键生成所有图表

```python
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from scipy import stats

# 读取数据
data = pd.read_csv('MetricsData.csv')

# 设置样式
sns.set_style("whitegrid")
plt.rcParams['font.size'] = 12

# 1. HV箱线图
plt.figure(figsize=(10, 6))
sns.boxplot(x='Algorithm', y='HV', data=data, palette='Set2')
sns.stripplot(x='Algorithm', y='HV', data=data, 
              color='black', alpha=0.3, size=3)
plt.title('Hypervolume Comparison')
plt.ylabel('Hypervolume')
plt.tight_layout()
plt.savefig('hv_boxplot.png', dpi=300)
plt.close()

# 2. IGD箱线图
plt.figure(figsize=(10, 6))
sns.boxplot(x='Algorithm', y='IGD', data=data, palette='Set2')
sns.stripplot(x='Algorithm', y='IGD', data=data, 
              color='black', alpha=0.3, size=3)
plt.title('IGD Comparison')
plt.ylabel('Inverted Generational Distance')
plt.tight_layout()
plt.savefig('igd_boxplot.png', dpi=300)
plt.close()

# 3. 组合图
fig, axes = plt.subplots(1, 2, figsize=(15, 6))
sns.boxplot(x='Algorithm', y='HV', data=data, ax=axes[0], palette='Set2')
axes[0].set_title('Hypervolume')
sns.boxplot(x='Algorithm', y='IGD', data=data, ax=axes[1], palette='Set2')
axes[1].set_title('IGD')
plt.tight_layout()
plt.savefig('combined_boxplot.png', dpi=300)
plt.close()

# 4. 统计检验
print("=== Wilcoxon检验 ===")
algorithms = ['NSGAII', 'MOEAD', 'MOGWO', 'SPEA2']
for i, alg1 in enumerate(algorithms):
    for alg2 in algorithms[i+1:]:
        data1 = data[data['Algorithm']==alg1]['HV']
        data2 = data[data['Algorithm']==alg2]['HV']
        _, p = stats.wilcoxon(data1, data2)
        sig = "**" if p < 0.01 else ("*" if p < 0.05 else "ns")
        print(f"{alg1} vs {alg2}: p={p:.6f} {sig}")

print("\n✅ 所有图表已生成！")
```
